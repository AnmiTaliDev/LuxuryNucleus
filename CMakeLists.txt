cmake_minimum_required(VERSION 4.3.1)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_BUILD_TYPE Release)
project(LuxuryNucleus LANGUAGES CXX ASM)

set(VALID_ARCHITECTURE "x86")
set(arch "" CACHE STRING "The target architecture.")
set(cross "" CACHE STRING "The cross compiler prefix (without minus at the end).")
if(CMAKE_CXX_COMPILER STREQUAL "g++")
    set(CMAKE_CXX_COMPILER "${cross}-${CMAKE_CXX_COMPILER}")
elseif(CMAKE_CXX_COMPILER STREQUAL "clang++")
    add_compile_options(--target=${cross})
endif()

if(NOT arch IN_LIST VALID_ARCHITECTURE)
    message(FATAL_ERROR "Please, choose the correct architecture: ${VALID_ARCHITECTURE}")
elseif(arch STREQUAL "x86")
    set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} -masm=intel")
    add_compile_options(-masm=intel -mno-red-zone)
endif()

set(LINKER_SCRIPT "${CMAKE_SOURCE_DIR}/arch/${arch}/script.ld")
add_compile_options(-Wall -Wextra -Werror -O0 -ffreestanding -fno-PIC -fno-signed-char -funsigned-char -nostdlib -m32)
add_link_options(-T${LINKER_SCRIPT} -z max-page-size=0 -nostdlib -no-pie -O0 -m32)
include_directories("${CMAKE_SOURCE_DIR}")

add_executable(LuxuryNucleus $<TARGET_OBJECTS:init_main>)
set_target_properties(LuxuryNucleus PROPERTIES LINK_DEPENDS ${LINKER_SCRIPT})

set_target_properties(LuxuryNucleus PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ISO/boot/")
add_custom_command(TARGET LuxuryNucleus POST_BUILD COMMAND grub-file --is-x86-multiboot2 $<TARGET_FILE:LuxuryNucleus>)
add_custom_command(TARGET LuxuryNucleus POST_BUILD COMMAND grub-mkrescue -o "${CMAKE_BINARY_DIR}/TEST.iso" "${CMAKE_BINARY_DIR}/ISO/")
add_custom_command(TARGET LuxuryNucleus POST_BUILD COMMAND qemu-system-x86_64 -enable-kvm -machine q35 -cpu host -m 4096 -cdrom "${CMAKE_BINARY_DIR}/TEST.iso")
add_custom_command(TARGET LuxuryNucleus POST_BUILD COMMAND echo ${CMAKE_CXX_FLAGS})

add_subdirectory(arch)
add_subdirectory(library)
add_subdirectory(drivers)
add_subdirectory(init)
add_subdirectory(mm)
add_subdirectory(video)
add_subdirectory(debug)
